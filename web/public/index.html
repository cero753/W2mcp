Or generate from the command line

      
# from a docs URL (LLM reads the HTML)
npx tsx src/cli.ts https://your-api.com/docs --out ./out/myapi --render

# from an OpenAPI / Swagger spec (JSON or YAML — no LLM)
npx tsx src/import-cli.ts https://your-api.com/openapi.json --out ./out/myapi

# verify it against the live API, then host it
npx tsx src/verify-cli.ts ./out/myapi
npx tsx src/gateway-cli.ts        # serves every server in registry.json
